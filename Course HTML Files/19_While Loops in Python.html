While Loop with Else Statement

num = 0

while(num < 5):
    print("The number is",num)
    num = num + 1

The pass Keyword

x = [1,2,3]
for item in x:
    pass

The continue Keyword

my_string = "HELLO"
for letter in my_string:
    if letter == "E":
        continue
    print(letter)

The break Keyword

x = 0
while x < 5:
    if x == 2:
        break
    print(x)
    x = x + 1
